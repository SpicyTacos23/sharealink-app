#!/bin/bash
set -e

MYSQL_HOST=${MYSQL_HOST:-mysql}
MYSQL_DATABASE=${MYSQL_DATABASE:-mediaBridge}
MYSQL_USER=${MYSQL_USER:-mediaTest}
MYSQL_PASSWORD=${MYSQL_PASSWORD:-Test123.}
REDIS_URL=${REDIS_URL:-redis://redis:6379}
MAX_RETRIES=30
SLEEP_SECONDS=2

check_mysql() {
    local tries=0
    echo "⏳ Esperando a MySQL..."
    until mariadb-admin ping -h "$MYSQL_HOST" -u"$MYSQL_USER" -p"$MYSQL_PASSWORD" --skip-ssl --silent; do
        tries=$((tries+1))
        if [ "$tries" -ge "$MAX_RETRIES" ]; then
            echo "❌ MySQL no responde"
            exit 1
        fi
        sleep "$SLEEP_SECONDS"
    done
    echo "✅ MySQL listo"
}

check_redis() {
    if command -v redis-cli >/dev/null 2>&1; then
        local redis_host redis_port
        redis_host=$(echo "$REDIS_URL" | sed -E 's#^[^:]+://([^:/]+).*#\1#')
        redis_port=$(echo "$REDIS_URL" | sed -E 's#^[^:]+://[^:/]+:([0-9]+).*#\1#; t; s#.*://[^:/]+.*#6379#')
        local tries=0
        echo "⏳ Esperando a Redis en $redis_host:$redis_port..."
        until redis-cli -h "$redis_host" -p "$redis_port" ping >/dev/null 2>&1; do
            tries=$((tries+1))
            if [ "$tries" -ge "$MAX_RETRIES" ]; then
                echo "❌ Redis no responde"
                exit 1
            fi
            sleep "$SLEEP_SECONDS"
        done
        echo "✅ Redis listo"
    else
        echo "⚠️ redis-cli no disponible, se omite."
    fi
}

check_mysql
check_redis

echo "📦 Generando migraciones..."
php bin/console doctrine:migrations:diff --no-interaction --allow-empty-diff || true

echo "📦 Ejecutando migraciones..."
php bin/console doctrine:migrations:migrate --no-interaction --allow-no-migration

echo "🌱 Cargando fixtures..."
php bin/console doctrine:fixtures:load --no-interaction --append

echo "🗑️ Limpiando caché..."
php bin/console cache:clear --no-interaction

echo "🚀 Arrancando servicios..."
exec /usr/bin/supervisord -c /etc/supervisor/conf.d/supervisord.conf
